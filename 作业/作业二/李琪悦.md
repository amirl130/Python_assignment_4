```python
import math
from functools import reduce
import itertools

# ============================================
# 一、迭代类
# ============================================

def sum_for(data):
    """for 循环累加"""
    total = 0
    for x in data:
        total += x
    return total

def sum_while(data):
    """while 循环累加"""
    total = 0
    i = 0
    while i < len(data):
        total += data[i]
        i += 1
    return total

def sum_listcomp(data):
    """列表推导式求和"""
    new_list = [x for x in data]
    return sum(new_list)

def sum_generator(data):
    """生成器表达式求和"""
    gen = (x for x in data)
    return sum(gen)

def sum_for_else(data):
    """for-else 语法求和"""
    total = 0
    for x in data:
        total += x
    else:
        return total

# ============================================
# 二、函数式
# ============================================

def sum_builtin(data):
    """内置 sum() 直接调用"""
    return sum(data)

def sum_reduce(data):
    """functools.reduce 归约"""
    if not data:
        return 0
    return reduce(lambda x, y: x + y, data)

def sum_map(data, func=None):
    """map + sum 组合"""
    if func is None:
        func = lambda x: x
    return sum(map(func, data))

def sum_accumulate(data):
    """itertools.accumulate 求和"""
    if not data:
        return 0
    return list(itertools.accumulate(data))[-1]

# ============================================
# 三、递归类
# ============================================

def sum_recursive(data):
    """直接递归求和"""
    if not data:
        return 0
    return data[0] + sum_recursive(data[1:])

def sum_tail_recursive(data, acc=0):
    """尾递归（累积器模式）"""
    if not data:
        return acc
    return sum_tail_recursive(data[1:], acc + data[0])

def sum_divide_conquer(data):
    """分治递归求和"""
    if len(data) == 0:
        return 0
    if len(data) == 1:
        return data[0]
    mid = len(data) // 2
    return sum_divide_conquer(data[:mid]) + sum_divide_conquer(data[mid:])

# ============================================
# 四、数值稳定类
# ============================================

def sum_fsum(data):
    """math.fsum 高精度求和"""
    return math.fsum(data)

def sum_kahan(data):
    """Kahan 补偿求和算法"""
    s = 0.0
    c = 0.0
    for x in data:
        y = x - c
        t = s + y
        c = (t - s) - y
        s = t
    return s

def sum_sorted(data):
    """排序后累加"""
    sorted_data = sorted(data)
    total = 0
    for x in sorted_data:
        total += x
    return total

def sum_neumaier(data):
    """Neumaier 改进版 Kahan 算法"""
    s = 0.0
    c = 0.0
    for x in data:
        t = s + x
        if abs(s) >= abs(x):
            c += (s - t) + x
        else:
            c += (x - t) + s
        s = t
    return s + c
