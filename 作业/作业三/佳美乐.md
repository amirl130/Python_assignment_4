```python
# ============================================
# 作业3：BRICS国家FDI与经济增长关系分析
# 成员C：佳美乐 — 热力图 + 3D回归曲面图
# ============================================

# 1. 导入库
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from mpl_toolkits.mplot3d import Axes3D

# 设置中文显示
plt.rcParams['font.sans-serif'] = ['SimHei', 'Arial']
plt.rcParams['axes.unicode_minus'] = False

# 2. 读取数据
df = pd.read_csv('brics_fdi_timeseries.csv')
print("数据加载成功！")
print(df.head())

# ============================================
# 可视化②：热力图（相关系数矩阵）
# ============================================

print("\n" + "="*50)
print("正在生成热力图...")
print("="*50)

# 选择数值列
numeric_cols = ['GDP_growth', 'FDI_inflow', 'GDP_percapita']
corr_matrix = df[numeric_cols].corr()

# 绘制热力图
plt.figure(figsize=(8, 6))
sns.heatmap(corr_matrix, 
            annot=True,           # 显示数值
            fmt='.2f',            # 保留2位小数
            cmap='RdBu_r',        # 红蓝配色
            center=0,
            square=True,
            linewidths=1,
            cbar_kws={"shrink": 0.8})
plt.title('图②：相关系数热力图\nBRICS国家（2010-2019）', fontsize=14)
plt.tight_layout()
plt.savefig('heatmap.png', dpi=300, bbox_inches='tight')
plt.show()

print("✅ 热力图已保存为 heatmap.png")

# ============================================
# 可视化③：3D回归曲面图
# ============================================

print("\n" + "="*50)
print("正在生成3D回归曲面图...")
print("="*50)

# 准备数据
X1 = df['FDI_inflow'].values          # FDI流入量
X2 = df['GDP_percapita'].values        # 人均GDP
Y = df['GDP_growth'].values            # GDP增长率

# 添加常数项 (X0 = 1)
X = np.column_stack((np.ones(len(X1)), X1, X2))

# OLS回归: Y = b0 + b1*X1 + b2*X2
coeffs, residuals, rank, s = np.linalg.lstsq(X, Y, rcond=None)
b0, b1, b2 = coeffs

print(f"\n回归方程: GDP增长率 = {b0:.3f} + {b1:.3f}×FDI + {b2:.6f}×人均GDP")

# 创建网格用于绘制曲面
x1_grid = np.linspace(X1.min(), X1.max(), 30)
x2_grid = np.linspace(X2.min(), X2.max(), 30)
X1_grid, X2_grid = np.meshgrid(x1_grid, x2_grid)
Y_grid = b0 + b1 * X1_grid + b2 * X2_grid

# 国家颜色映射
country_colors = {
    'China': 'red',
    'India': 'orange', 
    'Brazil': 'green',
    'South Africa': 'blue'
}

# 绘制3D图
fig = plt.figure(figsize=(14, 10))
ax = fig.add_subplot(111, projection='3d')

# 绘制散点（真实数据）
for country in df['Country'].unique():
    subset = df[df['Country'] == country]
    ax.scatter(subset['FDI_inflow'], 
               subset['GDP_percapita'], 
               subset['GDP_growth'],
               c=country_colors.get(country, 'gray'),
               s=80,
               label=country,
               edgecolors='black',
               linewidth=1,
               alpha=0.8)

# 绘制回归曲面
ax.plot_surface(X1_grid, X2_grid, Y_grid, 
                alpha=0.4, 
                color='gray',
                edgecolor='none')

# 设置标签
ax.set_xlabel('FDI流入量 (占GDP%)', fontsize=12)
ax.set_ylabel('人均GDP (PPP, 国际美元)', fontsize=12)
ax.set_zlabel('GDP增长率 (%)', fontsize=12)
ax.set_title('图③：3D回归曲面图\nFDI与人均GDP对GDP增长的联合影响', fontsize=14)

# 设置视角
ax.view_init(elev=25, azim=45)

# 图例
ax.legend(loc='upper left', fontsize=10)

plt.tight_layout()
plt.savefig('3d_regression.png', dpi=300, bbox_inches='tight')
plt.show()

print("✅ 3D曲面图已保存为 3d_regression.png")

# ============================================
# 输出相关系数矩阵
# ============================================
print("\n" + "="*50)
print("相关系数矩阵：")
print("="*50)
print(corr_matrix.round(3))

print("\n" + "="*50)
print("✅ 全部完成！")
print("生成文件：heatmap.png, 3d_regression.png")
print("="*50)
