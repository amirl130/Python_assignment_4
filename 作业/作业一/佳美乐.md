## 💻 我的代码

```python
import random

# 初始化变量
knowledge = 50
phy_health = 70
men_health = 70
credits = 0
total_gpa = 0
game_over = False

def calc_gpa(score):
    """分数转绩点"""
    if score >= 90:
        return 4.0
    elif score >= 80:
        return 3.0
    elif score >= 70:
        return 2.0
    elif score >= 60:
        return 1.0
    else:
        return 0.0

# 主循环：学年-学期双层嵌套
for year in range(1, 5):
    for sem in range(1, 3):
        if game_over:
            break
        
        print(f"\n=== 第{year}学年 第{sem}学期 ===")
        print(f"知识:{knowledge} 身体:{phy_health} 心理:{men_health}")
        print("选择: S(学习) P(娱乐) E(运动) X(独处) Y(社交)")
        
        choice = input("你的选择: ").upper().strip()
        
        if choice == 'S':
            knowledge += random.randint(5, 15)
            phy_health -= random.randint(3, 8)
            men_health += random.randint(1, 5)
            print("📚 学习！")
        elif choice == 'P':
            men_health += random.randint(5, 12)
            knowledge -= random.randint(1, 5)
            print("🎮 放松！")
        elif choice == 'E':
            phy_health += random.randint(4, 10)
            men_health += random.randint(1, 4)
            knowledge -= random.randint(1, 4)
            print("🏃 运动！")
        elif choice == 'X':
            men_health -= random.randint(5, 12)
            print("😔 独处")
        elif choice == 'Y':
            men_health += random.randint(4, 10)
            print("👥 社交")
        else:
            print("⚠️ 无效输入")
        
        # 边界控制
        knowledge = max(0, knowledge)
        phy_health = max(0, phy_health)
        men_health = max(0, men_health)
        
        # 健康检查
        if phy_health <= 25 or men_health <= 40:
            print("💀 健康崩溃！游戏结束")
            game_over = True
            break
    
    if game_over:
        break
    
    # 知识衰减机制（每学年末）
    decay = random.randint(3, 8)
    knowledge = max(0, knowledge - decay)
    print(f"\n📖 学年结束，遗忘知识 {decay} 点")
    
    # 考试
    if knowledge >= 80:
        exam_score = random.randint(85, 100)
    elif knowledge >= 60:
        exam_score = random.randint(70, 84)
    elif knowledge >= 40:
        exam_score = random.randint(55, 69)
    else:
        exam_score = random.randint(40, 54)
    
    gpa = calc_gpa(exam_score)
    total_gpa += gpa
    credits += random.randint(30, 45)
    
    print(f"📝 考试成绩: {exam_score} → 绩点: {gpa}")
    print(f"🎓 总学分: {credits}")
    
    # GPA淘汰检查
    if year >= 2:
        avg_gpa = total_gpa / (year * 2)
        if avg_gpa < 2.1:
            print(f"📉 平均绩点 {avg_gpa:.2f} < 2.1，淘汰")
            game_over = True
            break

# 结局判定
if not game_over:
    avg_gpa = total_gpa / 8
    print("\n" + "="*40)
    if credits >= 140 and avg_gpa >= 2.1:
        print("✅ 恭喜毕业！")
    else:
        print("❌ 未能毕业")
